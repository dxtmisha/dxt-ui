### `updateValue`

Event fires when the selected value changes programmatically (for example, via `previous()` / `next()` navigation) while `isSelectedByValue` is enabled.

**Parameters:**
- `value?: NumberOrStringOrBoolean` — new selected value

<Source
    code={`
<script setup lang="ts">
import { ref } from 'vue'
import type { NumberOrStringOrBoolean } from '@dxtmisha/functional'

const selected = ref('item1')
const menuItems = [
  { label: 'Menu Item 1', value: 'item1' },
  { label: 'Menu Item 2', value: 'item2' },
  { label: 'Menu Item 3', value: 'item3' }
]

const handleUpdateValue = (value?: NumberOrStringOrBoolean) => {
  console.log('New value:', value)
}
</script>

<template>
  <Menu
    v-model:selected="selected"
    :list="menuItems"
    is-selected-by-value
    @update-value="handleUpdateValue"
  >
    <template #control="{ next }">
      <button type="button" @click="next">Next Item</button>
    </template>
  </Menu>
</template>
  `}
    language="html"
/>
